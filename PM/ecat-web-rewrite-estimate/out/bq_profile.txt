--- org_feature_usage_report
CREATE VIEW `supercat-data-pipeline.mixpanel.org_feature_usage_report`
AS WITH user_metrics AS ( SELECT LOWER(COALESCE(current_organization_shortname, organization_shortname)) as org, COUNT(DISTINCT LOWER(username)) as total_users, COUNT(DISTINCT CASE WHEN time >= UNIX_SECONDS(TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 90 DAY)) THEN LOWER(username) END) as active_users, COUNT(DISTINCT CONCAT(LOWER(username), '_', DATE(TIMESTAMP_SECONDS(CAST(time AS INT64))))) as total_logins, COUNT(*) as total_events FROM `supercat-data-pipeline.WELD_RAW.mixpanel__events` WHERE COALESCE(current_organization_shortname, organization_shortname) IS NOT NULL AND username IS NOT NULL AND LOWER(username) NOT LIKE '%supercatsolutions.com%' AND LOWER(username) NOT LIKE '%test%' GROUP BY org ), event_counts AS ( SELECT LOWER(COALESCE(current_organization_shortname, organization_shortname)) as org, COUNTIF(event_name = 'product_search') as search_products, COUNTIF(event_name = 'filter_button_pressed') as filter_products, COUNTIF(event_name = 'product_sort_changed') as change_catalog_sort, COUNTIF(event_name = 'collection_search') as search_collections, COUNTIF(event_name = 'create_stack' AND (type = 'my_list' OR type IS NULL)) as create_my_list, COUNTIF(event_name = 'edit_stack') as edit_my_list, COUNTIF(event_name = 'view_stack') as view_my_list, COUNTIF(event_name = 'email_stack') as share_my_list, COUNTIF(event_name = 'create_stack' AND type = 'customer_product_list') as create_customer_product_list, COUNTIF(event_name = 'view_customer_on_order_items') as view_cust_product_list, COUNTIF(event_name = 'create_stack' AND type = 'maybe_list') as create_maybe_list, COUNTIF(event_name = 'item_email_drafted') as email_item_info, COUNTIF(event_name = 'pdf_catalog_generated') as create_pdf_catalog, COUNTIF(event_name = 'csv_report_generated') as export_data_to_csv, COUNTIF(event_name = 'generate_xlsx_catalog') as export_data_to_excel, COUNTIF(event_name = 'view_document') as view_library_entry, COUNTIF(event_name = 'document_email_drafted' AND SAFE_CAST(item_count AS INT64) = 1) as email_single_library_entry, COUNTIF(event_name = 'document_email_drafted' AND SAFE_CAST(item_count AS INT64) > 1) as email_multiple_library_entries, COUNTIF(event_name = 'pspdfkit_activated') as pdf_searches, COUNTIF(event_name = 'customer_selection') as select_a_customer, COUNTIF(event_name = 'customer_search') as search_for_customer, COUNTIF(event_name = 'show_customer_sales_setting_changed') as show_sales_in_catalog, COUNTIF(event_name = 'view_favorites') as view_cust_favorites, COUNTIF(event_name = 'view_customer_on_order_items') as view_cust_backorders, COUNTIF(event_name = 'view_customer_smart_picks') as view_smartpicks, COUNTIF(event_name = 'view_portal') as access_sales_portal, COUNTIF(event_name = 'view_customer_orders') as view_ipad_orders, COUNTIF(event_name = 'view_kit') as view_kit, COUNTIF(event_name = 'add_kit_to_order') as order_kit, COUNTIF(event_name = 'add_configured_item_to_order') as order_configured_item, COUNTIF(event_name = 'add_to_order_from_maybe_list') as order_from_maybe_list, COUNTIF(event_name = 'item_scanned') as scan_item_with_camera, COUNTIF(event_name = 'order_submitted') as submit_order, COUNTIF(event_name = 'view_customer_placements') as view_placements, COUNTIF(event_name = 'view_commitments') as view_commitments, COUNTIF(event_name = 'view_flipbook') as view_flipbook, COUNTIF(event_name = 'flipbook_add_to_list') as add_to_list_from_flipbook, COUNTIF(event_name = 'flipbook_add_to_order') as order_from_flipbook FROM `supercat-data-pipeline.WELD_RAW.mixpanel__events` WHERE COALESCE(current_organization_shortname, organization_shortname) IS NOT NULL GROUP BY org ), deduped_mapping AS ( SELECT LOWER(organization_shortname) AS org_shortname, ANY_VALUE(customer_name) AS customer_name, ANY_VALUE(hubspot_company_id) AS hubspot_company_id FROM `supercat-data-pipeline.mixpanel.organization_customer_mapping` GROUP BY LOWER(organization_shortname) ) SELECT e.org AS org_shortname, COALESCE(m.customer_name, e.org) AS org_name, m.hubspot_company_id, COALESCE(u.total_users, 0) AS total_users, COALESCE(u.active_users, 0) AS active_users, COALESCE(u.total_logins, 0) AS total_logins, COALESCE(u.total_events, 0) AS total_events, COALESCE(e.search_products, 0) AS search_products, COALESCE(e.filter_products, 0) AS filter_products, COALESCE(e.change_catalog_sort, 0) AS change_catalog_sort, COALESCE(e.search_collections, 0) AS search_collections, COALESCE(e.create_my_list, 0) AS create_my_list, COALESCE(e.edit_my_list, 0) AS edit_my_list, COALESCE(e.view_my_list, 0) AS view_my_list, COALESCE(e.share_my_list, 0) AS COALESCE(e.email_multiple_library_entries, 0) AS email_multiple_library_entries, COALESCE(e.pdf_searches, 0) AS pdf_searches, COALESCE(e.select_a_customer, 0) AS select_a_customer, COALESCE(e.search_for_customer, 0) AS search_for_customer, COALESCE(e.show_sales_in_catalog, 0) AS show_sales_in_catalog, COALESCE(e.view_cust_favorites, 0) AS view_cust_favorites, COALESCE(e.view_cust_backorders, 0) AS view_cust_backorders, COALESCE(e.view_smartpicks, 0) AS view_smartpicks, COALESCE(e.access_sales_portal, 0) AS access_sales_portal, COALESCE(e.view_ipad_orders, 0) AS view_ipad_orders, COALESCE(e.view_kit, 0) AS view_kit, COALESCE(e.order_kit, 0) AS order_kit, COALESCE(e.order_configured_item, 0) AS order_configured_item, COALESCE(e.order_from_maybe_list, 0) AS order_from_maybe_list, COALESCE(e.scan_item_with_camera, 0) AS scan_item_with_camera, COALESCE(e.submit_order, 0) AS submit_order, COALESCE(e.view_placements, 0) AS view_placements, COALESCE(e.view_commitments, 0) AS view_commitments, COALESCE(e.view_flipbook, 0) AS view_flipbook, COALESCE(e.add_to_list_from_flipbook, 0) AS add_to_list_from_flipbook, COALESCE(e.order_from_flipbook, 0) AS order_from_flipbook FROM event_counts e LEFT JOIN deduped_mapping m ON e.org = m.org_shortname LEFT JOIN user_metrics u ON e.org = u.org ORDER BY e.org;

--- user_feature_usage_report
CREATE VIEW `supercat-data-pipeline.mixpanel.user_feature_usage_report`
AS WITH user_events AS (
  SELECT 
    LOWER(COALESCE(current_organization_shortname, organization_shortname)) as org,
    LOWER(username) as username,
    COUNT(DISTINCT DATE(TIMESTAMP_SECONDS(CAST(time AS INT64)))) as days_active,
    COUNT(*) as total_events,
    MIN(TIMESTAMP_SECONDS(CAST(time AS INT64))) as first_event,
    MAX(TIMESTAMP_SECONDS(CAST(time AS INT64))) as last_event,
    COUNTIF(event_name = 'product_search') as search_products,
    COUNTIF(event_name = 'filter_button_pressed') as filter_products,
    COUNTIF(event_name = 'product_sort_changed') as change_catalog_sort,
    COUNTIF(event_name = 'collection_search') as search_collections,
    COUNTIF(event_name = 'create_stack' AND (type = 'my_list' OR type IS NULL)) as create_my_list,
    COUNTIF(event_name = 'edit_stack') as edit_my_list,
    COUNTIF(event_name = 'view_stack') as view_my_list,
    COUNTIF(event_name = 'email_stack') as share_my_list,
    COUNTIF(event_name = 'create_stack' AND type = 'customer_product_list') as create_customer_product_list,
    COUNTIF(event_name = 'view_customer_on_order_items') as view_cust_product_list,
    COUNTIF(event_name = 'create_stack' AND type = 'maybe_list') as create_maybe_list,
    COUNTIF(event_name = 'item_email_drafted') as email_item_info,
    COUNTIF(event_name = 'pdf_catalog_generated') as create_pdf_catalog,
    COUNTIF(event_name = 'csv_report_generated') as export_data_to_csv,
    COUNTIF(event_name = 'generate_xlsx_catalog') as export_data_to_excel,
    COUNTIF(event_name = 'view_document') as view_library_entry,
    COUNTIF(event_name = 'document_email_drafted' AND SAFE_CAST(item_count AS INT64) = 1) as email_single_library_entry,
    COUNTIF(event_name = 'document_email_drafted' AND SAFE_CAST(item_count AS INT64) > 1) as email_multiple_library_entries,
    COUNTIF(event_name = 'pspdfkit_activated') as pdf_searches,
    COUNTIF(event_name = 'customer_selection') as select_a_customer,
    COUNTIF(event_name = 'customer_search') as search_for_customer,
    COUNTIF(event_name = 'show_customer_sales_setting_changed') as show_sales_in_catalog,
    COUNTIF(event_name = 'view_favorites') as view_cust_favorites,
    COUNTIF(event_name = 'view_customer_on_order_items') as view_cust_backorders,
    COUNTIF(event_name = 'view_customer_smart_picks') as view_smartpicks,
    COUNTIF(event_name = 'view_portal') as access_sales_portal,
    COUNTIF(event_name = 'view_customer_orders') as view_ipad_orders,
    COUNTIF(event_name = 'view_kit') as view_kit,
    COUNTIF(event_name = 'add_kit_to_order') as order_kit,
    COUNTIF(event_name = 'add_configured_item_to_order') as order_configured_item,
    COUNTIF(event_name = 'add_to_order_from_maybe_list') as order_from_maybe_list,
    COUNTIF(event_name = 'item_scanned') as scan_item_with_camera,
    COUNTIF(event_name = 'order_submitted') as submit_order,
    COUNTIF(event_name = 'view_customer_placements') as view_placements,
    COUNTIF(event_name = 'view_commitments') as view_commitments,
    COUNTIF(event_name = 'view_flipbook') as view_flipbook,
    COUNTIF(event_name = 'flipbook_add_to_list') as add_to_list_from_flipbook,
    COUNTIF(event_name = 'flipbook_add_to_order') as order_from_flipbook
  FROM `supercat-data-pipeline.WELD_RAW.mixpanel__events`
  WHERE COALESCE(current_organization_shortname, organization_shortname) IS NOT NULL
    AND username IS NOT NULL
    AND LOWER(username) NOT LIKE '%supercatsolutions.com%'
    AND LOWER(username) NOT LIKE '%test%'
  GROUP BY org, username
)

SELECT * FROM user_events;
